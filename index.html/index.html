⌨️ Code Studio

ESP32 Blink Example

A simple example using GPIO 2. Connect an LED circuit correctly before testing.

#define LED_PIN 2

void setup() {
  pinMode(LED_PIN, OUTPUT);
}

void loop() {
  digitalWrite(LED_PIN, HIGH);
  delay(1000);

  digitalWrite(LED_PIN, LOW);
  delay(1000);
}
